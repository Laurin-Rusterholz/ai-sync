n vorher an Vorgaben eingetippt hat, soll nicht verloren gehen. Ist die
  // Notiz der Idee schon enthalten, passiert nichts (mehrfaches Uebernehmen
  // derselben Idee darf den Prompt nicht aufblaehen).
  function applyIdeaToPrompt(promptInput, ideaInput) {
    var prompt = Object.assign({}, asObject(promptInput));
    var idea = asObject(ideaInput);
    if (idea.areaName) prompt.areaName = idea.areaName;
    if (idea.title) prompt.moduleTitle = idea.title;
    var addition = cleanText(idea.notes, 4000);
    if (addition) {
      var existing = String(prompt.notes == null ? "" : prompt.notes);
      if (existing.indexOf(addition) === -1) {
        prompt.notes = existing.trim() ? existing.replace(/\s+$/, "") + "\n\n" + addition : addition;
      }
    }
    return prompt;
  }

  // Arbeit ist eine Aufgabe — und zwar dieselbe Aufgabenart wie in Quantus,

  var PREPARATION_STEPS = [
    "Quellen und Unterlagen zusammentragen",

    "Modul importieren und ersten Lerntag planen"

  function preparationId(areaName, moduleTitle) {
    return makeId("prep", String(areaName || "") + "-" + String(moduleTitle || ""));
  }

  function buildPreparation(input, now) {
    input = asObject(input);
    var areaName = cleanLine(input.areaName, 180) || "Karrierebereich";
    var moduleTitle = cleanLine(input.moduleTitle, 180) || "Neues Modul";
    var timestamp = now && !Number.isNaN(Date.parse(now)) ? new Date(now).toISOString() : new Date().toISOString();
    return {
      id: preparationId(areaName, moduleTitle),
      areaName: areaName,
      moduleTitle: moduleTitle,
      level: cleanLine(input.level, 180),
      dailyMinutes: Number(input.dailyMinutes) || 30,
      notes: cleanText(input.notes, 4000),
      targetDate: cleanLine(input.targetDate, 10) || null,
      steps: PREPARATION_STEPS.slice(),
      status: "open",
      taskRef: null,          // { source: "quantus-task", inboxId } — die echte Quantus-Aufgabe
      createdAt: timestamp,
      updatedAt: timestamp
    };
  }

  // Beschreibung der Quantus-Aufgabe. Bewusst vollstaendig, damit die Aufgabe
  // auch ausserhalb des Career Models fuer sich allein verstaendlich ist.
  function preparationTaskDescription(preparation) {
    var prep = asObject(preparation);
    // null bedeutet "diese Zeile entfaellt"; "" ist eine gewollte Leerzeile.
    return [
      "Modul fuer das Career Model vorbereiten.",
      "",
      "Bereich: " + (prep.areaName || "—"),
      "Modul: " + (prep.moduleTitle || "—"),
      prep.level ? "Niveau: " + prep.level : null,
      "Pensum: " + (prep.dailyMinutes || 30) + " Minuten pro Lerntag",
      prep.targetDate ? "Zieldatum: " + prep.targetDate : null,
      "",
      "Schritte:",
      asArray(prep.steps).map(function (step, index) { return (index + 1) + ". " + step; }).join("\n"),
      prep.notes ? "\nVorgaben:\n" + prep.notes : null,
      "",
      "Fertig ist die Vorbereitung, sobald das Modul im Career Model importiert ist."
    ].filter(function (line) { return line !== null; }).join("\n");
  }


  function moduleProgress(stateInput, moduleId) {
    var state = normalizeState(stateInput);
    var module = state.modules[moduleId];
    if (!module) return { completed: 0, total: 0, percent: 0, nextDayId: null };
    var order = asArray(module.dayOrder);
    var completedDays = asObject(state.progress[moduleId] && state.progress[moduleId].completedDays);
    var completed = order.filter(function (id) { return Boolean(completedDays[id]); }).length;
    var nextDayId = order.find(function (id) { return !completedDays[id]; }) || null;
    return { completed: completed, total: order.length, percent: order.length ? Math.round(completed / order.length * 100) : 0, nextDayId: nextDayId };
  }

  function nextSession(stateInput) {
    var state = normalizeState(stateInput);
    var modules = Object.keys(state.modules).map(function (id) { return state.modules[id]; })
      .filter(function (module) { return module && module.status !== "archived"; })
      .sort(function (a, b) {
        var aActive = a.status === "active" ? 1 : 0;
        var bActive = b.status === "active" ? 1 : 0;
        if (aActive !== bActive) return bActive - aActive;
        return String(b.updatedAt || b.importedAt || "").localeCompare(String(a.updatedAt || a.importedAt || ""));
      });
    for (var i = 0; i < modules.length; i += 1) {
      var module = modules[i];
      var stats = moduleProgress(state, module.id);
      if (!stats.nextDayId) continue;
      var day = asObject(module.days)[stats.nextDayId];
      if (!day) continue;
      return {
        module: module,
        area: state.areas[module.areaId] || null,
        day: day,
        progress: stats,
        completion: asObject(asObject(state.progress[module.id]).completedDays)[day.id] || null
      };
    }
    return null;
  }

  function mergePackage(stateInput, packageInput, now) {
    var validation = validatePackage(packageInput);
    if (!validation.ok) {
      var error = new Error("Career-Model-Datei ist ungültig.");
      error.validation = validation;
      throw error;
    }
    var pkg = validation.value;
    var state = normalizeState(deepClone(stateInput || {}));
    var timestamp = now && !Number.isNaN(Date.parse(now)) ? new Date(now).toISOString() : new Date().toISOString();
    var areaId = pkg.careerArea.id;
    var moduleId = pkg.module.id;
    var previousModule = state.modules[moduleId] || null;
    var previousAreaId = previousModule && previousModule.areaId;
    var previousProgress = asObject(state.progress[moduleId]);
    var dayOrder = pkg.module.days.map(function (day) { return day.id; });
    var days = {};
    pkg.module.days.forEach(function (day) { days[day.id] = day; });

    if (previousAreaId && previousAreaId !== areaId && state.areas[previousAreaId]) {
      state.areas[previousAreaId].moduleIds = asArray(state.areas[previousAreaId].moduleIds).filter(function (id) { return id !== moduleId; });
      state.areas[previousAreaId].updatedAt = timestamp;
    }

    var area = Object.assign({}, state.areas[areaId] || {}, pkg.careerArea, {
      id: areaId,
      moduleIds: Array.from(new Set(asArray(state.areas[areaId] && state.areas[areaId].moduleIds).concat(moduleId))),
      createdAt: state.areas[areaId] && state.areas[areaId].createdAt || timestamp,
      updatedAt: timestamp
    });
    state.areas[areaId] = area;

    state.modules[moduleId] = {
      id: moduleId,
      areaId: areaId,
      title: pkg.module.title,
      description: pkg.module.description,
      level: pkg.module.level,
      sourceDocuments: pkg.module.sourceDocuments,
      estimatedTotalMinutes: pkg.module.estimatedTotalMinutes,
      days: days,
      dayOrder: dayOrder,
      status: previousModule && previousModule.status || "active",
      importedAt: previousModule && previousModule.importedAt || timestamp,
      updatedAt: timestamp,
      sourceGeneratedAt: pkg.generatedAt,
      packageHash: stableHash(JSON.stringify(pkg))
    };

    var oldCompleted = asObject(previousProgress.completedDays);
    var completedDays = {};
    dayOrder.forEach(function (dayId) { if (oldCompleted[dayId]) completedDays[dayId] = oldCompleted[dayId]; });
    var stats = { completed: Object.keys(completedDays).length, total: dayOrder.length };
    state.progress[moduleId] = Object.assign({}, previousProgress, {
      moduleId: moduleId,
      completedDays: completedDays,
      currentDayId: dayOrder.find(function (id) { return !completedDays[id]; }) || null,
      startedAt: previousProgress.startedAt || timestamp,
      completedAt: stats.completed === stats.total && stats.total ? previousProgress.completedAt || timestamp : null,
      updatedAt: timestamp
    });

    state.meta.createdAt = state.meta.createdAt || timestamp;
    state.meta.updatedAt = timestamp;
    return state;
  }

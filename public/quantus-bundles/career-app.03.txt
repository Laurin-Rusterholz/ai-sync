ederholung nach Leitner");
    }

      '<span class="cm-pill">' + (card.kind === "question" ? "Kontrollfrage" : "Merksatz") + '</span></div>' +
      '<h2 class="cm-review-front">' + esc(card.front) + '</h2>' +
      (state.reviewRevealed
        ? '<div class="cm-review-back">' + esc(card.back || "(keine Antwort hinterlegt)") + '</div>' +
          '<div class="cm-form-actions">' +
          '<button class="cm-btn danger" data-cm-action="review-answer" data-card-id="' + attr(card.id) + '" data-correct="0">Nochmal ueben</button>' +
          '<button class="cm-btn green" data-cm-action="review-answer" data-card-id="' + attr(card.id) + '" data-correct="1">Gewusst</button></div>'
        : '<div class="cm-form-actions"><button class="cm-btn primary" data-cm-action="review-reveal" data-card-id="' + attr(card.id) + '">Antwort zeigen</button>' +
          '<button class="cm-btn" data-cm-action="review-skip" data-card-id="' + attr(card.id) + '">Ueberspringen</button></div>') +
      '<p class="cm-small cm-muted">' + (entry.seen ? esc(entry.seen + "× gesehen · " + (entry.correct || 0) + " richtig · " + (entry.wrong || 0) + " falsch") : "Neue Karte") + '</p>' +
      '</section>';
    return shell(body, "Wiederholen", due.length + " Karte(n) faellig");
  }

  // ── Fortschritt ──────────────────────────────────────────────────────────
  function progressView() {
    var stats = Core.progressStats(state.data);
    var review = Core.reviewStats(state.data);
    var areas = values(state.data.areas).sort(function (a, b) { return String(a.name).localeCompare(String(b.name), "de"); });
    var tiles = [
      ["Lerntage abgeschlossen", stats.doneDays + " / " + stats.totalDays],
      ["Gesamtfortschritt", stats.percent + " %"],
      ["Gelernte Zeit", Math.round(stats.minutes / 6) / 10 + " h"],
      ["Aktive Lerntage", String(stats.activeDays)],
      ["Serie", stats.streak + " Tag(e)"],
      ["Karten gemeistert", review.mastered + " / " + review.total]
    ];
    var content = '<div class="cm-plan-grid">' + tiles.map(function (tile) {
      return '<div class="cm-plan-cell"><small>' + esc(tile[0]) + '</small><strong>' + esc(tile[1]) + '</strong></div>';
    }).join("") + '</div>';
    content += '<div class="cm-section-head"><div><h2>Bereiche</h2><p>Wo du stehst, aufgeschluesselt nach Karrierebereich.</p></div></div>';
    content += areas.length ? areas.map(function (area) {
      var modules = asArray(area.moduleIds).map(function (id) { return state.data.modules[id]; }).filter(Boolean);
      return '<section class="cm-card pad"><div class="cm-section-head" style="margin-top:0"><div><h2>' + esc(area.name) +
        '</h2><p>' + modules.length + ' Modul(e)</p></div></div>' + modules.map(function (module) {

          var plan = Core.studyPlan(state.data, module.id);

            progressBar(moduleStats.percent) +

  // ── Modul vorbereiten ────────────────────────────────────────────────────

  }

  function validationHtml(validation) {
    if (!validation) return "";
    var rows = [];
    validation.errors.forEach(function (item) { rows.push('<div class="cm-message error"><strong>' + esc(item.path) + '</strong><br>' + esc(item.message) + '</div>'); });
    validation.warnings.forEach(function (item) { rows.push('<div class="cm-message warn"><strong>' + esc(item.path) + '</strong><br>' + esc(item.message) + '</div>'); });
    if (validation.ok) rows.unshift('<div class="cm-message ok"><strong>Datei ist importierbar.</strong><br>' + validation.metrics.days + ' Lerntage · ' + validation.metrics.totalMinutes + ' Minuten · Ø ' + validation.metrics.averageMinutes.toFixed(1) + ' Minuten pro Tag</div>');
    return '<div class="cm-validation">' + rows.join("") + '</div>';
  }

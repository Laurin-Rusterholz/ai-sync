ritt", stats.percent + " %"],

  // ── Modul vorbereiten ────────────────────────────────────────────────────
  // Die Vorbereitung eines Moduls ist Arbeit — und Arbeit ist in Quantus eine
  // Aufgabe. Deshalb legt jede Vorbereitung hier eine ECHTE Quantus-Aufgabe an
  // (ueber die Polaris-Inbox, die die Hauptapp ohnehin auswertet). Sie steht
  // danach unter „Aufgaben" und ist dieselbe Einheit wie jede andere Aufgabe —
  // keine Kopie und keine eigene Aufgabenart des Career Models.
  function preparationCard(prep) {
    var done = prep.status === "done";
    return '<article class="cm-card pad cm-prep' + (done ? " done" : "") + '">' +
      '<div class="cm-section-head" style="margin-top:0"><div><div class="cm-eyebrow">' + esc(prep.areaName) + '</div>' +
      '<h2>' + esc(prep.moduleTitle) + '</h2></div>' +
      (done ? '<span class="cm-pill green">Importiert</span>' : '<span class="cm-pill">In Vorbereitung</span>') + '</div>' +
      (prep.level ? '<p class="cm-muted">' + esc(prep.level) + '</p>' : "") +
      '<div class="cm-meta-row"><span class="cm-pill">◷ ' + esc(prep.dailyMinutes) + ' Min. pro Tag</span>' +
      (prep.targetDate ? '<span class="cm-pill">Ziel ' + esc(formatDate(prep.targetDate, { day: "2-digit", month: "2-digit", year: "numeric" })) + '</span>' : "") +

      '<button class="cm-btn danger" data-cm-action="prep-delete" data-prep-id="' + attr(prep.id) + '">Entfernen</button>' +
      '</div></article>';
  }

  function ideasSection() {
    var areaName = String(state.prompt.areaName || "").trim();
    var list = areaName ? Core.ideasForArea(state.data, areaName) : [];
    var others = Core.ideaAreas(state.data).filter(function (entry) {
      return entry.areaKey !== Core.areaKey(areaName);
    });
    var head = '<div class="cm-section-head"><div><h2>Themenideen</h2>' +
      '<p>' + (areaName ? 'Was du im Berufsfeld „' + esc(areaName) + '" irgendwann koennen willst.'
                        : 'Trage oben ein Berufsfeld ein, dann kannst du Ideen dazu sammeln.') + '</p></div>' +
      (list.length ? '<span class="cm-pill">' + list.filter(function (idea) { return idea.status !== "used"; }).length + ' offen</span>' : '') +
      '</div>';
    var form = areaName
      ? '<section class="cm-card pad"><div class="cm-form-grid">' +
        '<div class="cm-field"><label>Thema</label><input id="cmIdeaTitle" placeholder="z. B. Haftpflichtrecht"></div>' +
        '<div class="cm-field"><label>Notiz (optional)</label><input id="cmIdeaNotes" placeholder="Warum, welcher Schwerpunkt, welche Quelle"></div>' +
        '</div><div class="cm-form-actions"><button class="cm-btn primary" data-cm-action="idea-add">Idee sichern</button></div>' +
        (list.length ? '<div class="cm-idea-list">' + list.map(ideaRow).join("") + '</div>'
                     : '<p class="cm-small cm-muted">Noch keine Idee fuer dieses Berufsfeld.</p>') +
        '</section>'
      : '';
    // Ideen anderer Berufsfelder bleiben sichtbar — sonst verschwinden sie
    // lautlos, sobald oben ein anderes Feld eingetragen ist.
    var rest = others.length
      ? '<section class="cm-card pad"><h3 style="margin-top:0">Andere Berufsfelder</h3><div class="cm-idea-areas">' +
        others.map(function (entry) {
          return '<button class="cm-btn" data-cm-action="idea-area" data-area-name="' + attr(entry.areaName) + '">' +
            esc(entry.areaName) + ' <span class="cm-pill">' + entry.open + '/' + entry.total + '</span></button>';
        }).join("") + '</div></section>'
      : '';
    return head + form + rest;
  }

  function validationHtml(validation) {
    if (!validation) return "";
    var rows = [];
    validation.errors.forEach(function (item) { rows.push('<div class="cm-message error"><strong>' + esc(item.path) + '</strong><br>' + esc(item.message) + '</div>'); });
    validation.warnings.forEach(function (item) { rows.push('<div class="cm-message warn"><strong>' + esc(item.path) + '</strong><br>' + esc(item.message) + '</div>'); });
    if (validation.ok) rows.unshift('<div class="cm-message ok"><strong>Datei ist importierbar.</strong><br>' + validation.metrics.days + ' Lerntage · ' + validation.metrics.totalMinutes + ' Minuten · Ø ' + validation.metrics.averageMinutes.toFixed(1) + ' Minuten pro Tag</div>');
    return '<div class="cm-validation">' + rows.join("") + '</div>';
  }

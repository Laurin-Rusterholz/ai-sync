 function normalizeState(input) {
    var base = emptyState();
    input = asObject(input);
    base.schemaVersion = VERSION;
    base.areas = asObject(input.areas);
    base.modules = asObject(input.modules);
    base.progress = asObject(input.progress);
    base.reflections = asObject(input.reflections);
    base.preparations = asObject(input.preparations);
    base.plans = asObject(input.plans);
    base.review = asObject(input.review);
    base.ideas = asObject(input.ideas);
    base.settings = Object.assign(base.settings, asObject(input.settings));
    base.meta = Object.assign(base.meta, asObject(input.meta));
    return base;
  }

  // ── Lernplan ─────────────────────────────────────────────────────────────
  // Dieselbe Idee wie die BM-Vorbereitung: ein Zieldatum, daraus das noetige
  // Tagespensum und der Abgleich mit dem tatsaechlichen Stand. Der Unterschied
  // ist nur, WER rechnet — hier die App selbst statt eines n8n-Workflows.
  var LEITNER_DAYS = [1, 2, 4, 7, 14, 30];

  function addDaysKey(key, days) {
    var base = new Date(String(key || "").slice(0, 10) + "T12:00:00Z");
    if (Number.isNaN(base.getTime())) base = new Date();
    base.setUTCDate(base.getUTCDate() + Number(days || 0));
    return base.toISOString().slice(0, 10);
  }

  function daysBetweenKeys(from, to) {
    var a = new Date(String(from || "").slice(0, 10) + "T12:00:00Z");
    var b = new Date(String(to || "").slice(0, 10) + "T12:00:00Z");
    if (Number.isNaN(a.getTime()) || Number.isNaN(b.getTime())) return null;
    return Math.round((b - a) / 86400000);
  }

  // Liefert den Plan eines Moduls: verbleibende Tage, noetiges Pensum und ob man
  // vor- oder zurueckliegt. Ohne Zieldatum bleibt targetDate null und die App
  // zeigt nur den reinen Fortschritt.
  function studyPlan(stateInput, moduleId, todayKey) {
    var state = normalizeState(stateInput);
    var stats = moduleProgress(state, moduleId);
    var plan = asObject(state.plans[moduleId]);
    var today = todayKey || dateKey(new Date(), state.settings.timezone);
    var remaining = Math.max(0, stats.total - stats.completed);
    var result = {
      moduleId: moduleId,
      targetDate: plan.targetDate || null,
      startedAt: plan.startedAt || null,
      total: stats.total,
      completed: stats.completed,
      remaining: remaining,
      percent: stats.percent,
      daysLeft: null,
      perDay: null,
      expected: null,
      delta: null,
      status: "ok"
    };
    if (!plan.targetDate) return result;
    var daysLeft = daysBetweenKeys(today, plan.targetDate);
    result.daysLeft = daysLeft;
    if (daysLeft === null) return result;
    result.perDay = daysLeft > 0 ? Math.ceil(remaining / daysLeft) : remaining;
    // Erwarteter Stand: linear zwischen Start und Zieldatum.
    var start = plan.startedAt || today;
    var span = daysBetweenKeys(start, plan.targetDate);
    if (span && span > 0) {
      var elapsed = Math.max(0, Math.min(span, daysBetweenKeys(start, today)));
      result.expected = Math.round(stats.total * (elapsed / span));
      result.delta = stats.completed - result.expected;
    }
    if (daysLeft < 0 && remaining > 0) result.status = "overdue";
    else if (result.delta !== null && result.delta <= -2) result.status = "behind";
    else if (result.delta !== null && result.delta >= 2) result.status = "ahead";
    return result;
  }

  // ── Wiederholung (Leitner) ───────────────────────────────────────────────
  // Die Karten entstehen aus dem bereits importierten Stoff: Kontrollfragen und
  // Merksaetze jedes Lerntags. Es wird nichts generiert und nichts nachgeladen —
  // eine Karte existiert genau dann, wenn der Lerntag existiert.
  function cardId(moduleId, dayId, kind, index) {
    return moduleId + "::" + dayId + "::" + kind + index;
  }

  function reviewCards(stateInput, options) {
    var state = normalizeState(stateInput);
    options = asObject(options);
    var onlyCompleted = options.onlyCompleted !== false;   // Standard: nur gelernte Tage
    var cards = [];
    Object.keys(state.modules).forEach(function (moduleId) {
      if (options.moduleId && options.moduleId !== moduleId) return;
      var module = state.modules[moduleId];
      if (!module || module.status === "archived") return;
      var completed = asObject(asObject(state.progress[moduleId]).completedDays);
      asArray(module.dayOrder).forEach(function (dayId) {
        var day = asObject(module.days)[dayId];
        if (!day) return;
        if (onlyCompleted && !completed[dayId]) return;
        asArray(day.reviewQuestions).forEach(function (question, index) {
          if (!question || !question.question) return;
          cards.push({
            id: cardId(moduleId, dayId, "q", index),
            moduleId: moduleId, moduleTitle: module.title, dayId: dayId, dayTitle: day.title,
            kind: "question", front: question.question, back: question.answer || ""
          });
        });
        asArray(day.takeaways).forEach(function (item, index) {
          if (!item) return;
          cards.push({
            id: cardId(moduleId, dayId, "t", index),
            moduleId: moduleId, moduleTitle: module.title, dayId: dayId, dayTitle: day.title,
            kind: "takeaway", front: "Merksatz zu: " + (day.title || ""), back: String(item)
          });
        });
      });
    });
    return cards;
  }

  // Faellige Karten: alles, was noch nie gesehen wurde oder dessen Faelligkeit
  // erreicht ist. Neue Karten zuerst, danach die am laengsten ueberfaelligen.
  function dueCards(stateInput, options) {
    var state = normalizeState(stateInput);
    options = asObject(options);
    var today = options.todayKey || dateKey(new Date(), state.settings.timezone);
    return reviewCards(state, options).filter(function (card) {
      var entry = state.review[card.id];
      if (!entry || !entry.dueDate) return true;
      return String(entry.dueDate) <= today;
    }).sort(function (a, b) {
      var ea = state.review[a.id], eb = state.review[b.id];
      if (!ea && eb) return -1;
      if (ea && !eb) return 1;
      if (!ea && !eb) return 0;
      return String(ea.dueDate || "").localeCompare(String(eb.dueDate || ""));
    });
  }

  // Ergebnis einer Wiederholung: richtig schiebt eine Box weiter, falsch setzt
  // zurueck auf Box 1. Gibt den neuen Eintrag zurueck (der Aufrufer speichert).
  function applyReviewResult(entryInput, correct, todayKey) {
    var entry = asObject(entryInput);
    var box = Number(entry.box);
    if (!Number.isFinite(box) || box < 1) box = 1;
    box = correct ? Math.min(LEITNER_DAYS.length, box + 1) : 1;
    var today = todayKey || dateKey(new Date());
    return {
      box: box,
      dueDate: addDaysKey(today, LEITNER_DAYS[box - 1]),
      lastSeen: today,
      seen: Number(entry.seen || 0) + 1,
      correct: Number(entry.correct || 0) + (correct ? 1 : 0),
      wrong: Number(entry.wrong || 0) + (correct ? 0 : 1)
    };
  }

  // Beherrschungsgrad: Anteil der Karten, die mindestens Box 4 erreicht haben.
  function reviewStats(stateInput, options) {
    var state = normalizeState(stateInput);
    var cards = reviewCards(state, options);
    var due = dueCards(state, options).length;
    var mastered = 0, started = 0;
    cards.forEach(function (card) {
      var entry = state.review[card.id];
      if (!entry) return;
      started += 1;
      if (Number(entry.box || 1) >= 4) mastered += 1;
    });
    return {
      total: cards.length, due: due, started: started, mastered: mastered,
      percent: cards.length ? Math.round(mastered / cards.length * 100) : 0
    };
  }

  // ── Fortschritt ueber alle Module ────────────────────────────────────────
  function progressStats(stateInput) {
    var state = normalizeState(stateInput);
    var moduleIds = Object.keys(state.modules);
    var totalDays = 0, doneDays = 0, minutes = 0;
    var dates = {};
    moduleIds.forEach(function (moduleId) {
      var module = state.modules[moduleId];
      if (!module) return;
      var stats = moduleProgress(state, moduleId);
      totalDays += stats.total;
      doneDays += stats.completed;
      var completed = asObject(asObject(state.progress[moduleId]).completedDays);
      Object.keys(completed).forEach(function (dayId) {
        var entry = completed[dayId];
        if (!entry) return;
        minutes += Number(entry.actualMinutes || 0) || 0;
        var key = String(entry.completedAt || "").slice(0, 10);
        if (key) dates[key] = true;
      });
    });
    return {
      modules: moduleIds.length,
      areas: Object.keys(state.areas).length,
      totalDays: totalDays,
      doneDays: doneDays,
      percent: totalDays ? Math.round(doneDays / totalDays * 100) : 0,
      minutes: minutes,
      activeDays: Object.keys(dates).length,
      streak: streakLength(Object.keys(dates).sort(), dateKey(new Date(), state.settings.timezone))
    };
  }

  // Laufende Serie: heute oder gestern muss dabei sein, sonst ist sie gerissen.
  function streakLength(sortedDates, todayKey) {
    if (!sortedDates.length) return 0;
    var set = {};
    sortedDates.forEach(function (key) { set[key] = true; });
    var cursor = todayKey;
    if (!set[cursor]) {
      cursor = addDaysKey(todayKey, -1);
      if (!set[cursor]) return 0;
    }
    var count = 0;
    while (set[cursor]) { count += 1; cursor = addDaysKey(cursor, -1); }
    return count;
  }

  // ── Volltextsuche ueber alles Importierte ────────────────────────────────
  function searchContent(stateInput, query, limit) {
    var state = normalizeState(stateInput);
    var needle = cleanLine(query, 120).toLowerCase();
    if (needle.length < 2) return [];
    var hits = [];
    var max = Number(limit) > 0 ? Number(limit) : 60;
    function push(hit) { if (hits.length < max) hits.push(hit); }
    function matches(text) { return String(text || "").toLowerCase().indexOf(needle) > -1; }
    Object.keys(state.modules).forEach(function (moduleId) {
      var module = state.modules[moduleId];
      if (!module) return;
      asArray(module.dayOrder).forEach(function (dayId) {
        var day = asObject(module.days)[dayId];
        if (!day) return;
        var where = { moduleId: moduleId, moduleTitle: module.title, dayId: dayId, dayTitle: day.title, day: day.day };
        if (matches(day.title)) push(Object.assign({ kind: "Lerntag", text: day.title }, where));
        asArray(day.objectives).forEach(function (item) {
          if (matches(item)) push(Object.assign({ kind: "Lernziel", text: item }, where));
        });
        asArray(day.sections).forEach(function (section) {
          if (matches(section.title) || matches(section.text)) {
            push(Object.assign({ kind: "Lerninhalt", text: section.title || section.text }, where));
          }
          asArray(section.items).forEach(function (item) {
            if (matches(item)) push(Object.assign({ kind: "Lerninhalt", text: item }, where));
          });
        });
        asArray(day.reviewQuestions).forEach(function (question) {
          if (matches(question.question) || matches(question.answer)) {
            push(Object.assign({ kind: "Kontrollfrage", text: question.question }, where));
          }
        });
        asArray(day.takeaways).forEach(function (item) {
          if (matches(item)) push(Object.assign({ kind: "Merksatz", text: item }, where));
        });
        if (day.exercise && (matches(day.exercise.prompt) || matches(day.exercise.solution))) {
          push(Object.assign({ kind: "Praxisuebung", text: day.exercise.prompt }, where));
        }
      });
    });
    return hits;
  }

  // ── Themenideen je Beruf ─────────────────────────────────────────────────
  // Eine Idee ist ein Thema, das in einem Berufsfeld irgendwann ein Modul werden
  // soll — festgehalten in dem Moment, in dem es einem auffaellt, nicht erst wenn
  // man Zeit fuer die ganze Vorbereitung hat. Zugeordnet wird ueber den
  // normalisierten Berufsnamen, damit Ideen auch dann schon sammelbar sind, wenn
  // es fuer den Beruf noch gar keinen Bereich mit Modulen gibt.
  function areaKey(areaName) {
    return slugify(areaName, "beruf");
  }

  function buildIdea(input, now) {
    input = asObject(input);
    var areaName = cleanLine(input.areaName, 180) || "Karrierebereich";
    var title = cleanLine(input.title, 180);
    var timestamp = now && !Number.isNaN(Date.parse(now)) ? new Date(now).toISOString() : new Date().toISOString();
    return {
      id: makeId("idea", areaName + "-" + title),
      areaKey: areaKey(areaName),
      areaName: areaName,
      title: title,
      notes: cleanText(input.notes, 4000),
      status: "open",           // open | used — "used", sobald daraus ein Modul wurde
      usedAt: null,
      createdAt: timestamp,
      updatedAt: timestamp
    };
  }

  // Alle Ideen eines Berufs, offene zuerst, darin die neuesten oben.
  function ideasForArea(stateInput, areaName) {
    var state = normalizeState(stateInput);
    var key = areaKey(areaName);
    return Object.keys(state.ideas).map(function (id) { return state.ideas[id]; })
      .filter(function (idea) { return idea && idea.areaKey === key; })
      .sort(function (a, b) {
        var aOpen = a.status === "used" ? 1 : 0;
        var bOpen = b.status === "used" ? 1 : 0;
        if (aOpen !== bOpen) return aOpen - bOpen;
        return String(b.createdAt || "").localeCompare(String(a.createdAt || ""));
      });
  }

  // Alle Berufe, zu denen Ideen vorliegen — samt Zaehlern fuer die Uebersicht.
  function ideaAreas(stateInput) {
    var state = normalizeState(stateInput);
    var byKey = {};
    Object.keys(state.ideas).forEach(function (id) {
      var idea = state.ideas[id];
      if (!idea) return;
      var entry = byKey[idea.areaKey] || (byKey[idea.areaKey] = { areaKey: idea.areaKey, areaName: idea.areaName, total: 0, open: 0 });
      entry.total += 1;
      if (idea.status !== "used") entry.open += 1;
    });
    return Object.keys(byKey).map(function (key) { return byKey[key]; })
      .sort(function (a, b) { return String(a.areaName).localeCompare(String(b.areaName), "de"); });
  }

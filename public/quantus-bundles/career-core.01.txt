    var slug = cleanLine(value, 120).toLowerCase()
      .normalize("NFD").replace(/[\u0300-\u036f]/g, "")
      .replace(/ä/g, "ae").replace(/ö/g, "oe").replace(/ü/g, "ue").replace(/ß/g, "ss")
      .replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "").slice(0, 54);
    return slug || fallback || "eintrag";
  }

  function makeId(prefix, label) {
    var base = slugify(label, prefix || "item");
    var hash = stableHash(String(label || base)).slice(0, 7);
    return (prefix ? prefix + "-" : "") + base.slice(0, 48) + "-" + hash;
  }

  function stableHash(value) {
    var text = String(value || "");
    var hash = 2166136261;
    for (var i = 0; i < text.length; i += 1) {
      hash ^= text.charCodeAt(i);
      hash = Math.imul(hash, 16777619);
    }
    return (hash >>> 0).toString(36).padStart(7, "0");
  }

    options = options || {};
    if (!Array.isArray(value)) {
      pushError(result, path, "Muss eine Liste sein.");
      return [];
    }
    var output = value.map(function (item, index) {
      if (typeof item !== "string") {
        pushError(result, path + "[" + index + "]", "Muss Text sein.");
        return "";
      }
      var clean = cleanLine(item, options.maxLength || 700);
      if (!clean) pushError(result, path + "[" + index + "]", "Darf nicht leer sein.");
      if (unsafeMarkup(clean)) pushError(result, path + "[" + index + "]", "Enthält nicht erlaubtes HTML oder Script.");
      return clean;
    }).filter(Boolean);
    if (options.min != null && output.length < options.min) pushError(result, path, "Mindestens " + options.min + " Einträge erforderlich.");
    if (options.max != null && output.length > options.max) pushError(result, path, "Höchstens " + options.max + " Einträge erlaubt.");
    return output;
  }

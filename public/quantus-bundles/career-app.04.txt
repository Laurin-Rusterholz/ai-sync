  function render() {
    if (!state.user) {
      appRoot.innerHTML = loginView();
      return;
    }
    if (state.tab === "modules") appRoot.innerHTML = modulesView();
    else if (state.tab === "plan") appRoot.innerHTML = planView();
    else if (state.tab === "review") appRoot.innerHTML = reviewView();
    else if (state.tab === "progress") appRoot.innerHTML = progressView();
    else if (state.tab === "search") appRoot.innerHTML = searchView();
    else if (state.tab === "prepare") appRoot.innerHTML = prepareView();
    else if (state.tab === "import") appRoot.innerHTML = importView();
    else if (state.tab === "reflecta") appRoot.innerHTML = reflectaView();
    else appRoot.innerHTML = todayView();
  }

  function updateRoot(mutator) {
    if (!state.userRef) return Promise.reject(new Error("Keine Cloud-Verbindung."));
    setSync("loading", "Änderung wird gespeichert");
    return state.userRef.transaction(function (current) {
      var next = Core.normalizeState(current || {});
      var result = mutator(next) || next;
      result.meta = Object.assign({}, result.meta, { updatedAt: new Date().toISOString() });
      return result;
    }).then(function (result) {
      if (!result.committed) throw new Error("Die Änderung wurde nicht übernommen.");
      setSync("live", "Live auf allen Geräten");
      return result.snapshot.val();
    }).catch(function (error) {
      setSync("error", error.message || "Speichern fehlgeschlagen");
      throw error;
    });
  }

  function completeDay(form) {
    var data = new FormData(form);
    var moduleId = form.dataset.moduleId;
    var dayId = form.dataset.dayId;
    var completion = {
      completedAt: new Date().toISOString(),
      actualMinutes: Math.max(1, Number(data.get("actualMinutes")) || 1),
      confidence: String(data.get("confidence") || "mostly"),
      notes: String(data.get("notes") || "").trim()
    };
    return updateRoot(function (root) {
      var module = root.modules[moduleId];
      if (!module || !asObject(module.days)[dayId]) throw new Error("Lerntag nicht gefunden.");
      var progress = asObject(root.progress[moduleId]);
      var completedDays = Object.assign({}, asObject(progress.completedDays));
      completedDays[dayId] = completion;
      var nextId = asArray(module.dayOrder).find(function (id) { return !completedDays[id]; }) || null;
      root.progress[moduleId] = Object.assign({}, progress, {
        moduleId: moduleId,
        completedDays: completedDays,
        currentDayId: nextId,
        completedAt: nextId ? null : completion.completedAt,
        updatedAt: completion.completedAt
      });
      return root;
    }).then(function () {
      state.selectedModuleId = "";
      state.selectedDayId = "";
      toast("Lerneinheit abgeschlossen", "Fortschritt und Notiz sind live synchronisiert.", "ok");
    });
  }

  function reopenDay(moduleId, dayId) {
    return updateRoot(function (root) {
      var progress = asObject(root.progress[moduleId]);
      var completedDays = Object.assign({}, asObject(progress.completedDays));
      delete completedDays[dayId];
      root.progress[moduleId] = Object.assign({}, progress, { completedDays: completedDays, currentDayId: dayId, completedAt: null, updatedAt: new Date().toISOString() });
      return root;
    }).then(function () { toast("Lerntag wieder geöffnet", "Der Abschluss wurde entfernt.", "ok"); });
  }

  }

  function readImportFile(file) {
    state.importRaw = null;
    state.importValidation = null;
    if (!file) return render();
    if (file.size > Core.MAX_FILE_BYTES) {
      state.importValidation = { ok: false, errors: [{ path: "$", message: "Datei ist grösser als 5 MB." }], warnings: [], value: null };
      return render();
    }
    var reader = new FileReader();
    reader.onload = function () {
      try {
        var parsed = JSON.parse(String(reader.result || ""));
        state.importRaw = parsed;
        state.importValidation = Core.validatePackage(parsed);
      } catch (error) {
        state.importValidation = { ok: false, errors: [{ path: "$", message: "Ungültiges JSON: " + error.message }], warnings: [], value: null };
      }
      render();
    };
    reader.onerror = function () {
      state.importValidation = { ok: false, errors: [{ path: "$", message: "Datei konnte nicht gelesen werden." }], warnings: [], value: null };
      render();
    };
    reader.readAsText(file, "utf-8");
  }

  function importPackage() {
    if (!state.importRaw || !state.importValidation || !state.importValidation.ok) return Promise.reject(new Error("Die Datei ist nicht importierbar."));
    state.importing = true;
    render();
    var raw = state.importRaw;
    var moduleId = state.importValidation.value.module.id;
    var moduleTitle = state.importValidation.value.module.title;
    var firstDay = state.importValidation.value.module.days[0];
    return state.userRef.transaction(function (current) {
      return Core.mergePackage(current || {}, raw, new Date().toISOString());
    }).then(function (result) {
      if (!result.committed) throw new Error("Import wurde nicht übernommen.");
      state.importing = false;
      state.selectedModuleId = moduleId;
      state.selectedDayId = firstDay && firstDay.id || "";
      state.importRaw = null;
      state.importValidation = null;
      state.tab = "today";
      toast("Modul importiert", "Alle Lerntage sind sofort auf Tablet und Laptop verfügbar.", "ok");
      render();
      // Passende Vorbereitung samt Quantus-Aufgabe abschliessen.
      closeMatchingPreparation(moduleTitle);
    }).catch(function (error) {
      state.importing = false;
      render();
      throw error;
    });
  }

  function setModuleStatus(moduleId, status) {
    return updateRoot(function (root) {
      if (!root.modules[moduleId]) throw new Error("Modul nicht gefunden.");
      if (status === "active") {
        Object.keys(root.modules).forEach(function (id) {
          if (root.modules[id].areaId === root.modules[moduleId].areaId && root.modules[id].status === "active") root.modules[id].status = "ready";
        });
      }
      root.modules[moduleId].status = status;
      root.modules[moduleId].updatedAt = new Date().toISOString();
      return root;
    });
  }

  // ── Bruecke zu den Quantus-Aufgaben ──────────────────────────────────────
  // Quantus wertet polaris/inbox/<typ>/<id> aus und legt daraus ueber sein
  // eigenes createEntity() echte Entitaeten an. Genau diesen Weg nutzt das
  // Career Model: die Vorbereitungsaufgabe wird dadurch eine ganz normale
  // Quantus-Aufgabe, mit Aktivitaet und Undo, sichtbar unter „Aufgaben".
  // Kein Zugriff auf den Quantus-Datenstand, keine zweite Aufgabenart.
  //
  // Wichtig fuer spaetere Aenderungen: Quantus markiert einen verarbeiteten
  // Eintrag mit processedAt und ignoriert ihn danach. Ein Update schreibt den
  // Eintrag deshalb komplett neu (ohne processedAt) unter DERSELBEN inboxId —
  // Quantus findet die zugehoerige Aufgabe ueber seinen Index wieder.
  function pushQuantusTask(inboxId, payload, op) {
    if (!state.db) return Promise.reject(new Error("Keine Cloud-Verbindung."));
    var stamp = Date.now();
    return state.db.ref("polaris/inbox/task/" + inboxId).set(Object.assign({
      op: op || "create",
      source: "career-model",
      ts: stamp,
      updatedAt: stamp
    }, payload));
  }

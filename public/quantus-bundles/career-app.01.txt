  function esc(value) {
    return String(value == null ? "" : value)
      .replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;").replace(/'/g, "&#039;");
  }

  function asArray(value) { return Array.isArray(value) ? value : []; }
  function asObject(value) { return value && typeof value === "object" && !Array.isArray(value) ? value : {}; }
  function values(map) { return Object.keys(asObject(map)).map(function (key) { return map[key]; }).filter(Boolean); }

  function formatDate(value, options) {
    if (!value) return "";
    var date = value instanceof Date ? value : new Date(String(value).length === 10 ? value + "T12:00:00" : value);
    if (Number.isNaN(date.getTime())) return String(value);
    return new Intl.DateTimeFormat("de-CH", options || { weekday: "short", day: "2-digit", month: "long", year: "numeric" }).format(date);
  }

  function saveCache() {
    if (!state.user) return;
    try {
      global.localStorage.setItem(CACHE_PREFIX + state.user.uid, JSON.stringify({ data: state.data, savedAt: new Date().toISOString() }));
      global.localStorage.setItem(LAST_USER_KEY, state.user.uid);
    } catch (_) {}
  }

  function loadCache(uid) {
    try {
      var parsed = JSON.parse(global.localStorage.getItem(CACHE_PREFIX + uid) || "null");
      if (parsed && parsed.data) state.data = Core.normalizeState(parsed.data);
    } catch (_) {}
  }

  function setSync(status, message) {
    state.syncStatus = status;
    state.syncMessage = message || "";
  }

  function detach() {
    if (state.userRef && state.listener) state.userRef.off("value", state.listener);
    state.userRef = null;
    state.listener = null;
  }

  function attachUser(user) {
    detach();
    state.user = user || null;
    state.authReady = true;
    state.remoteReady = false;
    if (!user) {
      state.data = Core.emptyState();
      setSync("signed-out", "Nicht angemeldet");
      render();
      return;
    }
    loadCache(user.uid);
    setSync("loading", "Career Model wird synchronisiert");
    render();
    state.userRef = state.db.ref("careerModel/users/" + user.uid);
    state.listener = function (snapshot) {
      state.data = Core.normalizeState(snapshot.val() || {});
      state.remoteReady = true;
      setSync("live", "Live auf allen Geräten");
      saveCache();
      render();
    };
    state.userRef.on("value", state.listener, function (error) {
      setSync("error", error.message || "Synchronisation fehlgeschlagen");
      render();
    });
  }

  function initFirebase() {
    try {
      state.firebaseApp = global.firebase.apps.length ? global.firebase.app() : global.firebase.initializeApp(FIREBASE_CONFIG);
      state.auth = state.firebaseApp.auth();
      state.db = state.firebaseApp.database(RTDB_URL);
      state.auth.setPersistence(global.firebase.auth.Auth.Persistence.LOCAL).catch(function () {});
      state.auth.onAuthStateChanged(attachUser, function (error) {
        state.authReady = true;
        setSync("error", error.message || "Anmeldung fehlgeschlagen");
        render();
      });
    } catch (error) {
      state.authReady = true;
      setSync("error", error.message || "Firebase konnte nicht gestartet werden");
      render();
    }
  }

    if (node && node.getAttribute && node.getAttribute("href")) return;
    global.location.assign(QUANTUS_HOME_URL);
  }

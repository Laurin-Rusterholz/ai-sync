  // ── Lernplan ─────────────────────────────────────────────────────────────
  // Das Gegenstueck zum BM-Lernplan: ein Zieldatum je Modul, daraus rechnet die
  // App das noetige Tagespensum und den Abgleich mit dem tatsaechlichen Stand.
  // Der Unterschied zu BM ist nur, dass hier nichts von aussen geliefert wird.
  var PLAN_LABEL = {
    ok: '<span class="cm-pill green">Im Plan</span>',
    ahead: '<span class="cm-pill green">Vorsprung</span>',
    behind: '<span class="cm-pill warn">Rueckstand</span>',
    overdue: '<span class="cm-pill danger">Zieldatum verstrichen</span>'
  };

  function planCard(module) {
    var plan = Core.studyPlan(state.data, module.id);
    var area = state.data.areas[module.areaId];
    var rows = [
      ["Lerntage", plan.completed + " von " + plan.total],
      ["Offen", String(plan.remaining)],
      ["Verbleibende Tage", plan.daysLeft === null ? "—" : String(plan.daysLeft)],
      ["Noetiges Pensum", plan.perDay === null ? "—" : plan.perDay + " Lerntag(e) pro Tag"],
      ["Erwarteter Stand", plan.expected === null ? "—" : plan.expected + " Lerntage"],
      ["Abweichung", plan.delta === null ? "—" : (plan.delta > 0 ? "+" : "") + plan.delta + " Lerntage"]
    ];
    return '<section class="cm-card pad"><div class="cm-section-head" style="margin-top:0"><div>' +
      '<div class="cm-eyebrow">' + esc(area && area.name || "Karrierebereich") + '</div>' +
      '<h2>' + esc(module.title) + '</h2></div>' + (PLAN_LABEL[plan.status] || "") + '</div>' +
      progressBar(plan.percent) +
      '<div class="cm-plan-grid">' + rows.map(function (row) {
        return '<div class="cm-plan-cell"><small>' + esc(row[0]) + '</small><strong>' + esc(row[1]) + '</strong></div>';
      }).join("") + '</div>' +
      '<div class="cm-form-actions"><label class="cm-field" style="flex:1"><span class="cm-small">Zieldatum</span>' +
      '<input type="date" data-cm-plan="' + attr(module.id) + '" value="' + attr(plan.targetDate || "") + '"></label>' +
      '<button class="cm-btn" data-cm-action="plan-clear" data-module-id="' + attr(module.id) + '">Zieldatum entfernen</button>' +
      '</div></section>';
  }

  // ── Wiederholen (Leitner) ────────────────────────────────────────────────
  // Karten entstehen aus den Kontrollfragen und Merksaetzen abgeschlossener
  // Lerntage. Es wird nichts generiert: gibt es den Lerntag nicht, gibt es die
  // Karte nicht.
  function reviewView() {
    var options = { moduleId: state.reviewModuleId || undefined };
    var stats = Core.reviewStats(state.data, options);
    var due = Core.dueCards(state.data, options);
    var modules = values(state.data.modules).filter(function (module) { return module && module.status !== "archived"; });
    var filter = '<div class="cm-form-actions"><label class="cm-field" style="flex:1"><span class="cm-small">Modul</span>' +
      '<select data-cm-review-filter><option value="">Alle Module</option>' + modules.map(function (module) {
        return '<option value="' + attr(module.id) + '"' + (state.reviewModuleId === module.id ? " selected" : "") + '>' +
          esc(module.title) + '</option>';
      }).join("") + '</select></label></div>';

    var head = '<div class="cm-plan-grid">' +
      '<div class="cm-plan-cell"><small>Faellig</small><strong>' + stats.due + '</strong></div>' +
      '<div class="cm-plan-cell"><small>Karten gesamt</small><strong>' + stats.total + '</strong></div>' +
      '<div class="cm-plan-cell"><small>Gemeistert</small><strong>' + stats.mastered + '</strong></div>' +
      '<div class="cm-plan-cell"><small>Beherrschung</small><strong>' + stats.percent + ' %</strong></div>' +
      '<div class="cm-plan-cell"><small>Diese Sitzung</small><strong>' + state.reviewDone + '</strong></div>' +
      '</div>';

    if (!stats.total) {
      return shell('<section class="cm-card pad">' + filter + '</section>' +
        '<div class="cm-empty"><div class="cm-empty-icon">↻</div><h2>Noch nichts zu wiederholen</h2>' +
        '<p>Karten entstehen aus den Kontrollfragen und Merksaetzen der Lerntage, die du abgeschlossen hast. ' +
        'Schliesse deine erste Lerneinheit ab — danach steht sie hier zur Wiederholung bereit.</p>' +
        '<button class="cm-btn primary" data-cm-action="tab" data-tab="today">Zur Lerneinheit</button></div>',
        "Wiederholen", "Verteilte Wiederholung nach Leitner");
    }
    if (!due.length) {
      return shell('<section class="cm-card pad">' + filter + head + '</section>' +
        '<div class="cm-empty"><div class="cm-empty-icon">✓</div><h2>Fuer heute alles wiederholt</h2>' +
        '<p>Die naechsten Karten werden nach ihrem Leitner-Fach wieder faellig — nach 1, 2, 4, 7, 14 und 30 Tagen.</p></div>',
        "Wiederholen", "Verteilte Wiederholung nach Leitner");
    }

    var card = due.find(function (entry) { return entry.id === state.reviewCardId; }) || due[0];
    var entry = asObject(state.data.review[card.id]);
    var box = Number(entry.box || 1);
    var body = '<section class="cm-card pad">' + filter + head + '</section>' +
      '<section class="cm-card pad cm-review-card">' +
      '<div class="cm-meta-row"><span class="cm-pill accent">Fach ' + box + ' von ' + Core.LEITNER_DAYS.length + '</span>' +
      '<span class="cm-pill">▦ ' + esc(card.moduleTitle) + '</span>' +
      '<span class="cm-pill">' + esc(card.dayTitle || "") + '</span>' +
      '<span class="cm-pill">' + (card.kind === "question" ? "Kontrollfrage" : "Merksatz") + '</span></div>' +
      '<h2 class="cm-review-front">' + esc(card.front) + '</h2>' +
      (state.reviewRevealed
        ? '<div class="cm-review-back">' + esc(card.back || "(keine Antwort hinterlegt)") + '</div>' +
          '<div class="cm-form-actions">' +
          '<button class="cm-btn danger" data-cm-action="review-answer" data-card-id="' + attr(card.id) + '" data-correct="0">Nochmal ueben</button>' +
          '<button class="cm-btn green" data-cm-action="review-answer" data-card-id="' + attr(card.id) + '" data-correct="1">Gewusst</button></div>'
        : '<div class="cm-form-actions"><button class="cm-btn primary" data-cm-action="review-reveal" data-card-id="' + attr(card.id) + '">Antwort zeigen</button>' +
          '<button class="cm-btn" data-cm-action="review-skip" data-card-id="' + attr(card.id) + '">Ueberspringen</button></div>') +
      '<p class="cm-small cm-muted">' + (entry.seen ? esc(entry.seen + "× gesehen · " + (entry.correct || 0) + " richtig · " + (entry.wrong || 0) + " falsch") : "Neue Karte") + '</p>' +
      '</section>';
    return shell(body, "Wiederholen", due.length + " Karte(n) faellig");
  }

day, progress: Core.moduleProgress(state.data, module.id), completion: asObject(asObject(state.data.progress[module.id]).completedDays)[day.id] || null };
    }
    return Core.nextSession(state.data);
  }

  function syncDeviceContext(session) {
    if (!global.QuantusDeviceSync) return;
    var moduleId = session && session.module && session.module.id || "";
    var dayId = session && session.day && session.day.id || "";
    var query = moduleId ? "?module=" + encodeURIComponent(moduleId) + (dayId ? "&day=" + encodeURIComponent(dayId) : "") : "";
    global.QuantusDeviceSync.setContext({
      kind: "career",
      route: "career",
      collection: "careerModel",
      entityId: dayId || moduleId,
      title: session ? session.module.title + " · " + session.day.title : "Career Model",
      url: DESKTOP_ORIGIN + "/career-model.html" + query,
      sourceOrigin: global.location.origin
    });
  }

  function progressBar(percent) {
    return '<div class="cm-progress-track"><i style="width:' + Math.max(0, Math.min(100, Number(percent) || 0)) + '%"></i></div>';
  }

  function renderSection(section) {
    var type = section.type || "paragraph";
    return '<article class="cm-card cm-learning-section cm-type-' + attr(type) + '">' +
      (section.title ? '<h3>' + esc(section.title) + '</h3>' : '') +
      (section.text ? '<p>' + esc(section.text) + '</p>' : '') +
      (asArray(section.items).length ? '<ul>' + section.items.map(function (item) { return '<li>' + esc(item) + '</li>'; }).join("") + '</ul>' : '') +
      (asArray(section.sourceRefs).length ? '<div class="cm-source-refs">' + section.sourceRefs.map(function (ref) { return '<span>' + esc(ref) + '</span>'; }).join("") + '</div>' : '') +
    '</article>';
  }
